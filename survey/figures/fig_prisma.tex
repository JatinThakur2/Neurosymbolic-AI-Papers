% identification via databases
\node[box] (srcA) {\textbf{Source A} curated corpus~\corpuscite{}\\ $n=\cnt{corpusrecords}$ records\\ 7 venues, title keywords, 2020--2026};
\node[box, right=5mm of srcA] (srcB) {\textbf{Source B} proceedings metadata\\ $n=\cnt{sourceb}$ accepted papers with abstracts\\ stage-1 query: $n=\cnt{queryhits}$ retrieved\\ stage-2 query: $n=\cnt{searchtwo}$ further records};
\node[side, right=5mm of srcB] (dup) {Removed before screening:\\ duplicates in Source A $n=\cnt{corpusdup}$\\ in both sources (merged) $n=\cnt{overlap}$};
\node[box, below=7mm of $(srcA.south)!0.5!(srcB.south)$] (union) {\textbf{Records screened} (title + abstract)\\ stage 1: $n=\cnt{union}$; stage 2: $n=\cnt{searchtwo}$\\ two independent LLM screeners;\\ \cnt{adjunion} + \cnt{adjsearchtwo} sent to adjudication};
\node[side, right=5mm of union, xshift=18mm] (ex) {Excluded $n=\cnt{unionexcl}+\cnt{searchtwoexcl}$\\ (reasons pooled over all routes:\\ E1 \cnt{exclE1}, E2 \cnt{exclE2}, E3 \cnt{exclE3},\\ E4 \cnt{exclE4}, E5 \cnt{exclE5})};
\node[box, below=7mm of union] (inu) {Included from database searches\\ $n=\cnt{unionincl}+\cnt{searchtwoincl}$};
% other methods
\node[oth, below=7mm of srcA, xshift=-46mm] (snow) {\textbf{Backward snowballing}\\ references of \cnt{snowseeds} included studies\\ candidates screened $n=\cnt{snowcand}$};
\node[oth, below=3mm of snow] (aud) {\textbf{Recall audit} of non-retrieved\\ Source B records (stratified, double-\\screened): $n=\cnt{auditscreened}+\cnt{auditextn}$};
\node[oth, below=3mm of aud] (ki) {\textbf{Known-item check}: \cnt{kiref} references\\ of an independent survey; unscreened\\ ones screened: $n=\cnt{kiscreened}$};
\node[oth, below=3mm of ki] (oi) {Included from other methods\\ snowball $n=\cnt{snowincl}$, audit $n=\cnt{auditinclall}$,\\ known-item $n=\cnt{kiincl}$};
\node[inc, below=8mm of inu] (all) {\textbf{Studies included in the review}\\ $n=\cnt{included}$, double-coded by two LLM coders;\\ \cnt{adjcoding} adjudicated};
\draw[arr] (srcA.south) -- ++(0,-2mm) -| ([xshift=-6mm]union.north);

\node[stage, minimum width=12mm] at ($(all.west)+(-3.5mm,0)$) {Included};
